(function () {
    var GAP = 12;
    var nav = document.getElementById('rg-nav');
    var spacer = document.getElementById('rg-nav-spacer');
    var btn = document.getElementById('rg-mobile-menu-btn');
    var menu = document.getElementById('rg-mobile-menu');
    var icon = document.getElementById('rg-mobile-menu-icon');

    function syncOffset() {
        if (!nav) return;
        var offset = Math.ceil(nav.getBoundingClientRect().bottom) + GAP;
        document.documentElement.style.setProperty('--rg-nav-offset', offset + 'px');
        if (spacer) spacer.style.height = offset + 'px';
    }

    function onScroll() {
        if (!nav) return;
        nav.classList.toggle('rg-nav-scrolled', window.scrollY > 12);
    }

    function normalizePath(pathname) {
        var p = (pathname || '/').replace(/\/+$/, '');
        return p === '' ? '/' : p;
    }

    function syncActiveNav() {
        var path = normalizePath(window.location.pathname);
        var hash = (window.location.hash || '').replace(/^#/, '');
        var links = document.querySelectorAll('[data-rg-nav]');
        var activeKey = '';

        if (path.indexOf('about-us') !== -1) {
            activeKey = hash === 'pricing' ? 'pricing' : 'feature';
        } else if (path.indexOf('product') !== -1) {
            activeKey = 'product';
        } else if (path.indexOf('blog') !== -1) {
            activeKey = 'blog';
        } else if (path === '/') {
            activeKey = hash === 'katalog-tema' ? 'tema' : 'home';
        }

        links.forEach(function (link) {
            var key = link.getAttribute('data-rg-nav');
            link.classList.toggle('is-active', key === activeKey);
        });
    }

    function setMenuOpen(open) {
        if (!menu || !btn) return;
        menu.classList.toggle('hidden', !open);
        if (nav) nav.classList.toggle('rg-nav-open', open);
        btn.setAttribute('aria-expanded', open ? 'true' : 'false');
        if (icon) {
            icon.classList.toggle('fa-bars', !open);
            icon.classList.toggle('fa-times', open);
        }
        document.body.style.overflow = open ? 'hidden' : '';
        syncOffset();
    }

    syncOffset();
    onScroll();
    syncActiveNav();
    window.addEventListener('load', syncOffset);
    window.addEventListener('resize', syncOffset);
    window.addEventListener('scroll', onScroll, { passive: true });
    window.addEventListener('hashchange', syncActiveNav);

    if (typeof ResizeObserver !== 'undefined' && nav) {
        new ResizeObserver(syncOffset).observe(nav);
    }

    if (btn && menu) {
        btn.addEventListener('click', function () {
            setMenuOpen(menu.classList.contains('hidden'));
        });
        menu.querySelectorAll('a').forEach(function (link) {
            link.addEventListener('click', function () { setMenuOpen(false); });
        });
    }
})();

(function () {

    function syncOffset() {
        if (!nav) return;
        var offset = Math.ceil(nav.getBoundingClientRect().bottom) + GAP;
        document.documentElement.style.setProperty('--rg-nav-offset', offset + 'px');
        if (spacer) spacer.style.height = offset + 'px';

    function onScroll() {
        if (!nav) return;
        nav.classList.toggle('rg-nav-scrolled', window.scrollY > 12);
    }

    function normalizePath(pathname) {
        var p = (pathname || '/').replace(/\/+$/, '');
        return p === '' ? '/' : p;
    }

    function syncActiveNav() {
        var path = normalizePath(window.location.pathname);
        var hash = (window.location.hash || '').replace(/^#/, '');
        var links = document.querySelectorAll('[data-rg-nav]');
        var activeKey = '';

        if (path.indexOf('about-us') !== -1) {
            activeKey = hash === 'pricing' ? 'pricing' : 'feature';
        } else if (path.indexOf('product') !== -1) {
            activeKey = 'product';
        } else if (path.indexOf('blog') !== -1) {
            activeKey = 'blog';
        } else if (path === '/') {
            activeKey = hash === 'katalog-tema' ? 'tema' : 'home';
        }

        links.forEach(function (link) {
            var key = link.getAttribute('data-rg-nav');
            link.classList.toggle('is-active', key === activeKey);
        });
    }

    function setMenuOpen(open) {
        if (!menu || !btn) return;
        menu.classList.toggle('hidden', !open);
        if (nav) nav.classList.toggle('rg-nav-open', open);

        if (icon) {
            icon.classList.toggle('fa-bars', !open);
            icon.classList.toggle('fa-times', open);
        }
        document.body.style.overflow = open ? 'hidden' : '';
        syncOffset();
    }

    syncOffset();
    onScroll();
    syncActiveNav();

(function () {
    var reduceMotion = window.matchMedia ? window.matchMedia('(prefers-reduced-motion: reduce)') : null;
    var selector = [
        '[class*=\"animate-\"]',
        '.blob',
        '.fade-in',
        '.still-track',
        '.cover-slide-img',
        '.pc-hero__glow-orb'
    ].join(',');
    var nodes = [];
    var observer = null;

    function setPaused(el, paused) {
        if (!el || !el.style) return;
        el.style.animationPlayState = paused ? 'paused' : 'running';
    }

    function pauseByVisibility() {
        var hidden = document.hidden || (reduceMotion && reduceMotion.matches);
        for (var i = 0; i < nodes.length; i++) {
            if (hidden) {
                setPaused(nodes[i], true);
            } else if (!observer) {
                setPaused(nodes[i], false);
            } else {
                setPaused(nodes[i], nodes[i].__ajOffscreen === true);
            }
        }
    }

    function bindObserver() {
        if (!('IntersectionObserver' in window)) return;
        observer = new IntersectionObserver(function (entries) {
            entries.forEach(function (entry) {
                entry.target.__ajOffscreen = !entry.isIntersecting;
                setPaused(entry.target, document.hidden || entry.target.__ajOffscreen || (reduceMotion && reduceMotion.matches));
            });
        }, { threshold: 0.05 });
        for (var i = 0; i < nodes.length; i++) observer.observe(nodes[i]);
    }

    function refreshMotionNodes() {
        if (observer) observer.disconnect();
        observer = null;
        nodes = Array.prototype.slice.call(document.querySelectorAll(selector));
        if (!nodes.length) return;
        if (!(reduceMotion && reduceMotion.matches)) bindObserver();
        pauseByVisibility();
    }

    document.addEventListener('visibilitychange', pauseByVisibility);
    document.addEventListener('mitra:ready', refreshMotionNodes);
    if (reduceMotion && typeof reduceMotion.addEventListener === 'function') {
        reduceMotion.addEventListener('change', refreshMotionNodes);

    } else {
        refreshMotionNodes();
    }
})();
